(*Motor State Logic | Input and Output Structure*)
VAR
	Motor_In_Out : Motor_Variable; (*Variable to read input and control the outputs for motor*)
	StateMachineTest : State_Machine; (*State machine for machine logic flow*)
	MpAxisBasic_0 : MpAxisBasic;
	MpAxisBasic_1 : MpAxisBasic;
	MpAxisBasic_2 : MpAxisBasic;
	AxisParameters_0 : MpAxisBasicParType := (Direction:=mcDIR_SHORTEST_WAY,Homing:=(Mode:=mcHOMING_DIRECT,Position:=5),Stop:=(Deceleration:=5));
	AxisParameters_1 : MpAxisBasicParType := (Direction:=mcDIR_SHORTEST_WAY,Homing:=(Mode:=mcHOMING_DIRECT,Position:=5),Stop:=(Deceleration:=5));
	AxisParameters_2 : MpAxisBasicParType := (Direction:=mcDIR_SHORTEST_WAY,Homing:=(Mode:=mcHOMING_DIRECT,Position:=5),Stop:=(Deceleration:=5));
	iVa_MachineControl : MachineControl; (*Machine Control*)
END_VAR
VAR CONSTANT
	MIN_STOP_LIMIT : LREAL := 5;
	MAX_STOP_LIMIT : LREAL := 95;
END_VAR
VAR
	iVa_OldPos : REAL;
END_VAR

TYPE
	Motor_Variable : 	STRUCT  (*Motor Input and Output Structure*)
		Motor_In : Motor_In; (*Motor Input structure*)
		Motor_Out : ARRAY[1..3]OF Motor_Out; (*Motor Output structure*)
	END_STRUCT;
	Motor_In : 	STRUCT  (*Motor Input variables*)
		Position : REAL := 50.0;
		MovePos : BOOL;
		Home : BOOL;
		Stop : BOOL;
		ErrorReset : BOOL;
		PowerOn : BOOL;
	END_STRUCT;
	Motor_Out : 	STRUCT  (*Motor Output Variables*)
		Position : LREAL;
		Error : BOOL;
		HomeDone : BOOL;
		PoweredOn : BOOL;
		Velocity : REAL;
		ErrorGain : INT;
		Brake_Engage_Disengage : BOOL;
	END_STRUCT;
	State_Machine : 
		( (*State machine for motor encoder reading*)
		MotorError,
		PowerOn,
		Home,
		MotorMove,
		TestExtEncoder,
		STO_Error
		);
	MachineControl : 	STRUCT 
		Machine_Status : STRING[80];
		PowerOn_Home : BOOL;
		Move_Position : BOOL;
		Power_Failure : BOOL;
	END_STRUCT;
END_TYPE


PROGRAM _INIT    
	AxisParameters_0.Position 		:= AxisParameters_1.Position 	 := AxisParameters_2.Position		:= 50.0;
	AxisParameters_0.Velocity 		:= AxisParameters_1.Velocity 	 := AxisParameters_2.Velocity		:= 10.0;
	AxisParameters_0.Acceleration 	:= AxisParameters_1.Acceleration := AxisParameters_2.Acceleration	:= 50.0;
	AxisParameters_0.Deceleration 	:= AxisParameters_1.Deceleration := AxisParameters_1.Deceleration	:= 50.0;
	
	MpAxisBasic_0.MpLink 	 		:= ADR(gAxis_1);
	MpAxisBasic_0.Enable 	 		:= TRUE;
	MpAxisBasic_0.Parameters 		:= ADR(AxisParameters_0);
	
	MpAxisBasic_1.MpLink 	 		:= ADR(gAxis_2);
	MpAxisBasic_1.Enable 	 		:= TRUE;
	MpAxisBasic_1.Parameters 		:= ADR(AxisParameters_1);

	MpAxisBasic_2.MpLink 			:= ADR(gAxis_3);
	MpAxisBasic_2.Enable 			:= TRUE;
	MpAxisBasic_2.Parameters 		:= ADR(AxisParameters_2);
END_PROGRAM

PROGRAM _CYCLIC
	MpAxisBasic_0.Power 		:= MpAxisBasic_1.Power 			:= MpAxisBasic_2.Power 			:=  Motor_In_Out.Motor_In.PowerOn;
	MpAxisBasic_0.Home 			:= MpAxisBasic_1.Home 			:= MpAxisBasic_2.Home 			:=  Motor_In_Out.Motor_In.Home;
	MpAxisBasic_0.MoveAbsolute 	:= MpAxisBasic_1.MoveAbsolute	:= MpAxisBasic_2.MoveAbsolute	:=  Motor_In_Out.Motor_In.MovePos;
	MpAxisBasic_0.ErrorReset 	:= MpAxisBasic_1.ErrorReset 	:= MpAxisBasic_2.ErrorReset 	:=  Motor_In_Out.Motor_In.ErrorReset;
	MpAxisBasic_0.Stop 			:= MpAxisBasic_1.Stop 	 		:= MpAxisBasic_2.Stop		 	:=  Motor_In_Out.Motor_In.Stop;
	AxisParameters_0.Position	:= AxisParameters_1.Position 	:= AxisParameters_2.Position	:=  Motor_In_Out.Motor_In.Position;
	
	Motor_In_Out.Motor_Out[1].Error		:= MpAxisBasic_0.Error;
	Motor_In_Out.Motor_Out[1].HomeDone	:= MpAxisBasic_0.IsHomed;
	Motor_In_Out.Motor_Out[1].Position	:= MpAxisBasic_0.Position;
	Motor_In_Out.Motor_Out[1].PoweredOn	:= MpAxisBasic_0.PowerOn;
	Motor_In_Out.Motor_Out[1].Velocity	:= MpAxisBasic_0.Velocity;
	
	Motor_In_Out.Motor_Out[2].Error		:= MpAxisBasic_1.Error;
	Motor_In_Out.Motor_Out[2].HomeDone	:= MpAxisBasic_1.IsHomed;
	Motor_In_Out.Motor_Out[2].Position	:= MpAxisBasic_1.Position;
	Motor_In_Out.Motor_Out[2].PoweredOn := MpAxisBasic_1.PowerOn;
	Motor_In_Out.Motor_Out[2].Velocity	:= MpAxisBasic_1.Velocity;
	
	Motor_In_Out.Motor_Out[3].Error		:= MpAxisBasic_2.Error;
	Motor_In_Out.Motor_Out[3].HomeDone	:= MpAxisBasic_2.IsHomed;
	Motor_In_Out.Motor_Out[3].Position	:= MpAxisBasic_2.Position;
	Motor_In_Out.Motor_Out[3].PoweredOn := MpAxisBasic_2.PowerOn;
	Motor_In_Out.Motor_Out[3].Velocity	:= MpAxisBasic_2.Velocity;
	
	//Error handling logic
	IF (Motor_In_Out.Motor_Out[1].Error = TRUE OR Motor_In_Out.Motor_Out[2].Error = TRUE OR Motor_In_Out.Motor_Out[3].Error = TRUE) OR iVa_MachineControl.Power_Failure  THEN
		Motor_In_Out.Motor_In.MovePos						:= FALSE;
		Motor_In_Out.Motor_Out[1].Brake_Engage_Disengage	:= TRUE;
		Motor_In_Out.Motor_Out[2].Brake_Engage_Disengage	:= TRUE;
		Motor_In_Out.Motor_Out[3].Brake_Engage_Disengage	:= TRUE;
		StateMachineTest									:= STO_Error;
	END_IF;
		
	//Logic to run motor and test external encoder
	CASE StateMachineTest OF
		MotorError:
			iVa_MachineControl.Machine_Status	:= 'Machine in Error';
			iVa_MachineControl.Move_Position	:= FALSE;
			Motor_In_Out.Motor_In.MovePos		:= FALSE;
			iVa_MachineControl.PowerOn_Home 	:= FALSE;
				IF (Motor_In_Out.Motor_Out[1].Error = FALSE AND Motor_In_Out.Motor_Out[2].Error = FALSE AND Motor_In_Out.Motor_Out[3].Error = FALSE) THEN
					StateMachineTest	:= PowerOn;
				END_IF;
			
		PowerOn:
			iVa_MachineControl.Machine_Status  := 'Waiting to power on and Home';
			
			IF (Motor_In_Out.Motor_Out[1].Error = FALSE AND Motor_In_Out.Motor_Out[2].Error = FALSE AND Motor_In_Out.Motor_Out[3].Error = FALSE) AND
				(Motor_In_Out.Motor_Out[1].PoweredOn = FALSE AND Motor_In_Out.Motor_Out[2].PoweredOn = FALSE AND Motor_In_Out.Motor_Out[3].PoweredOn = FALSE) AND 
				iVa_MachineControl.PowerOn_Home = TRUE THEN
					Motor_In_Out.Motor_In.PowerOn	:= TRUE;
					StateMachineTest				:= Home;
			ELSIF
				(Motor_In_Out.Motor_Out[1].Error = FALSE AND Motor_In_Out.Motor_Out[2].Error = FALSE AND Motor_In_Out.Motor_Out[3].Error = FALSE) AND
				(Motor_In_Out.Motor_Out[1].PoweredOn = TRUE AND Motor_In_Out.Motor_Out[2].PoweredOn = TRUE AND Motor_In_Out.Motor_Out[3].PoweredOn = TRUE) AND 
				iVa_MachineControl.PowerOn_Home = TRUE THEN
					StateMachineTest	:= Home;
			END_IF;
	
		Home:
			IF (Motor_In_Out.Motor_Out[1].Error = FALSE AND Motor_In_Out.Motor_Out[2].Error = FALSE AND Motor_In_Out.Motor_Out[3].Error = FALSE) AND
				(Motor_In_Out.Motor_Out[1].PoweredOn = TRUE AND Motor_In_Out.Motor_Out[2].PoweredOn = TRUE AND Motor_In_Out.Motor_Out[3].PoweredOn = TRUE) 
				(*AND (Motor_In_Out.Motor_Out[1].HomeDone = FALSE AND Motor_In_Out.Motor_Out[2].HomeDone = FALSE AND Motor_In_Out.Motor_Out[3].HomeDone = FALSE)*) THEN
				Motor_In_Out.Motor_In.Home	:= TRUE;
					StateMachineTest			:= MotorMove;
//			ELSIF (Motor_In_Out.Motor_Out[1].Error = FALSE AND Motor_In_Out.Motor_Out[2].Error = FALSE AND Motor_In_Out.Motor_Out[3].Error = FALSE) AND
//				(Motor_In_Out.Motor_Out[1].PoweredOn = TRUE AND Motor_In_Out.Motor_Out[2].PoweredOn = TRUE AND Motor_In_Out.Motor_Out[3].PoweredOn = TRUE) AND 
//				(Motor_In_Out.Motor_Out[1].HomeDone = TRUE AND Motor_In_Out.Motor_Out[2].HomeDone = TRUE AND Motor_In_Out.Motor_Out[3].HomeDone = TRUE) THEN
//					StateMachineTest			:= MotorMove;
			END_IF;
			
		MotorMove:
			iVa_MachineControl.Machine_Status  := 'Waiting to Start the Test';
			
			IF (Motor_In_Out.Motor_Out[1].Error = FALSE AND Motor_In_Out.Motor_Out[2].Error = FALSE AND Motor_In_Out.Motor_Out[3].Error = FALSE) AND
				(Motor_In_Out.Motor_Out[1].PoweredOn = TRUE AND Motor_In_Out.Motor_Out[2].PoweredOn = TRUE AND Motor_In_Out.Motor_Out[3].PoweredOn = TRUE) AND
				(Motor_In_Out.Motor_Out[1].HomeDone = TRUE AND Motor_In_Out.Motor_Out[2].HomeDone = TRUE AND Motor_In_Out.Motor_Out[3].HomeDone = TRUE) AND
				iVa_MachineControl.Move_Position = TRUE THEN
					Motor_In_Out.Motor_In.MovePos	:= TRUE;
					StateMachineTest				:= TestExtEncoder;
			END_IF;
			
		TestExtEncoder:
			iVa_MachineControl.Machine_Status  := 'Machine Moving';
			
			IF iVa_MachineControl.Move_Position = FALSE OR (MpAxisBasic_0.MoveDone AND MpAxisBasic_1.MoveDone AND MpAxisBasic_2.MoveDone) THEN
				Motor_In_Out.Motor_In.MovePos		:= FALSE;
				iVa_MachineControl.Move_Position	:= FALSE;
				StateMachineTest				:= MotorMove;
			ELSIF ((Motor_In_Out.Motor_Out[1].Position >= MAX_STOP_LIMIT) OR (Motor_In_Out.Motor_Out[2].Position >= MAX_STOP_LIMIT) OR (Motor_In_Out.Motor_Out[3].Position >= MAX_STOP_LIMIT))  OR
				((Motor_In_Out.Motor_Out[1].Position < MIN_STOP_LIMIT) OR (Motor_In_Out.Motor_Out[2].Position < MIN_STOP_LIMIT) OR (Motor_In_Out.Motor_Out[3].Position < MIN_STOP_LIMIT)) OR
				((Motor_In_Out.Motor_Out[1].Velocity) <> (Motor_In_Out.Motor_Out[1].Velocity + Motor_In_Out.Motor_Out[1].ErrorGain)) OR
				((Motor_In_Out.Motor_Out[2].Velocity) <> (Motor_In_Out.Motor_Out[2].Velocity + Motor_In_Out.Motor_Out[2].ErrorGain))THEN
				//Motor_In_Out.Motor_In.Stop		:= TRUE;
				Motor_In_Out.Motor_In.MovePos						:= FALSE;
				Motor_In_Out.Motor_Out[1].Brake_Engage_Disengage	:= TRUE;
				Motor_In_Out.Motor_Out[2].Brake_Engage_Disengage	:= TRUE;
				Motor_In_Out.Motor_Out[3].Brake_Engage_Disengage	:= TRUE;
				StateMachineTest									:= STO_Error;
			END_IF;
		
		STO_Error:
			iVa_MachineControl.Machine_Status  	:= 'STO Error/Power Failure - Safe Stop (Brakes Engaged)';
			iVa_MachineControl.Move_Position	:= FALSE;
			iVa_MachineControl.PowerOn_Home 	:= FALSE;
			Motor_In_Out.Motor_In.Home			:= FALSE;
			Motor_In_Out.Motor_In.PowerOn		:= FALSE;
			IF (*MpAxisBasic_1.Stopped = TRUE AND MpAxisBasic_2.Stopped = TRUE AND*) Motor_In_Out.Motor_In.ErrorReset THEN
				//Motor_In_Out.Motor_In.Stop 		:= FALSE;
				//Motor_In_Out.Motor_In.MovePos	:= FALSE;
				Motor_In_Out.Motor_Out[1].Brake_Engage_Disengage	:= FALSE;
				Motor_In_Out.Motor_Out[2].Brake_Engage_Disengage	:= FALSE;
				Motor_In_Out.Motor_Out[3].Brake_Engage_Disengage	:= FALSE;
				Motor_In_Out.Motor_In.ErrorReset					:= FALSE;
				StateMachineTest		   							:= MotorError;
			END_IF;
		
	END_CASE;
	
	(* call function block for first axis *)
	MpAxisBasic_0;
	
	(* call function block for second axis *)
	MpAxisBasic_1;
	
	(* call function block for third axis *)
	MpAxisBasic_2;
	
	IF Motor_In_Out.Motor_In.Position <> iVa_OldPos THEN
		iVa_OldPos := Motor_In_Out.Motor_In.Position;
		MpAxisBasic_0.Update := MpAxisBasic_1.Update := MpAxisBasic_2.Update := TRUE;
	ELSIF ((MpAxisBasic_0.UpdateDone AND MpAxisBasic_1.UpdateDone AND MpAxisBasic_2.UpdateDone) = TRUE) THEN
		MpAxisBasic_0.Update := MpAxisBasic_1.Update := MpAxisBasic_2.Update := FALSE;
	END_IF;
	
END_PROGRAM

PROGRAM _EXIT
	(* Insert code here *)
	MpAxisBasic_0.Enable := FALSE;
	MpAxisBasic_0.Power  := FALSE;
	MpAxisBasic_0.Home   := FALSE;
	MpAxisBasic_0.MoveAbsolute := FALSE;
	MpAxisBasic_0.MoveAdditive := FALSE;
	MpAxisBasic_0.MoveVelocity := FALSE;
	MpAxisBasic_0.Stop         := FALSE;
	MpAxisBasic_0.ErrorReset   := FALSE;
	MpAxisBasic_0();
	
	MpAxisBasic_1.Enable := FALSE;
	MpAxisBasic_1.Power  := FALSE;
	MpAxisBasic_1.Home   := FALSE;
	MpAxisBasic_1.MoveAbsolute := FALSE;
	MpAxisBasic_1.MoveAdditive := FALSE;
	MpAxisBasic_1.MoveVelocity := FALSE;
	MpAxisBasic_1.Stop         := FALSE;
	MpAxisBasic_1.ErrorReset   := FALSE;
	MpAxisBasic_1();
	
	MpAxisBasic_2.Enable := FALSE;
	MpAxisBasic_2.Power  := FALSE;
	MpAxisBasic_2.Home   := FALSE;
	MpAxisBasic_2.MoveAbsolute := FALSE;
	MpAxisBasic_2.MoveAdditive := FALSE;
	MpAxisBasic_2.MoveVelocity := FALSE;
	MpAxisBasic_2.Stop         := FALSE;
	MpAxisBasic_2.ErrorReset   := FALSE;
	MpAxisBasic_2();
END_PROGRAM
